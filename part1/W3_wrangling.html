Setup: Example Data

All examples in this document use the patients dataset created below.

library(tidyverse)

set.seed(42)

n <- 40

patients <- tibble(
  pt_id     = paste0("P", sprintf("%03d", 1:n)),
  gender    = sample(c("M", "F"), n, replace = TRUE),
  height    = round(rnorm(n, mean = 168, sd = 10)),
  weight    = round(rnorm(n, mean = 70,  sd = 15)),
  Age       = round(runif(n, min = 30, max = 72)),
  Diagnosis = sample(c("Diabetes", "Hypertension", "Healthy"),
                     n, replace = TRUE, prob = c(0.35, 0.35, 0.30)),
  disease   = Diagnosis,
  HbA1c     = round(
    case_when(
      Diagnosis == "Diabetes"     ~ rnorm(n, mean = 8.0, sd = 1.0),
      Diagnosis == "Hypertension" ~ rnorm(n, mean = 5.8, sd = 0.7),
      TRUE                        ~ rnorm(n, mean = 5.2, sd = 0.5)
    ), 1)
)

Reading / Writing Files

# Working directory
getwd()
setwd("your/path/")

# readr (tidyverse): returns a tibble
patients <- read_csv('data/patients.csv')    # comma-separated
patients <- read_tsv('data/patients.txt')    # tab-separated

# base R: returns a data.frame (requires separator and header specification)
patients <- read.table('data/patients.csv', sep = ",", header = T)

# save
write_csv(patients, file = "data/patients2.csv")
write.table(patients, file = "data/patients2.txt",
            sep = "\t", quote = F, row.names = F, col.names = T)

2. Visualization: ggplot2

ggplot2 is built on three components:

  1. data: a data frame
  2. aes: x-axis, y-axis, color, fill, group, etc.
  3. geom: the type of plot
dataframe %>%
  ggplot(aes(x = xx, y = yy)) +
  geom_function() +
  ...

Aesthetics

Property Description Example
color / fill Color "red", "#RRGGBB"
linetype Line type "solid", "dashed", "dotted"
size Point / text size in mm
linewidth Line width in mm
shape Point shape integer 0–25
alpha Transparency 0 (transparent) – 1 (opaque)
Tip

Inside aes(): maps to a variable automatically → aes(color = Diagnosis)

Inside geom_: fixed value → geom_bar(color = 'black')

① Bar Plot: frequency of categorical data

patients %>%
  ggplot(aes(x = Diagnosis)) +
  geom_bar(fill = "steelblue", width = 0.6) +
  labs(title = "Number of Patients by Diagnosis",
       x = "Diagnosis", y = "Count") +
  theme_classic()

② Histogram: distribution of a continuous variable

patients %>%
  ggplot(aes(x = Age)) +
  geom_histogram(binwidth = 10, fill = "lightgrey", color = "black") +
  labs(title = "Age Distribution", x = "Age", y = "Count") +
  theme_classic()

③ Box Plot: comparing continuous values across groups

patients %>%
  ggplot(aes(x = Diagnosis, y = HbA1c, fill = Diagnosis)) +
  geom_boxplot() +
  labs(title = "HbA1c by Diagnosis", x = "Diagnosis", y = "HbA1c (%)") +
  theme_classic()

④ Scatter Plot: relationship between two continuous variables

patients %>%
  ggplot(aes(x = Age, y = HbA1c)) +
  geom_point(size = 3, color = 'darkblue') +
  geom_smooth(method = "lm", formula = y ~ x, se = TRUE) +
  labs(title = "Age vs HbA1c", x = "Age", y = "HbA1c (%)") +
  theme_classic()

⑤ Line Plot: trends over time

bp_long %>%
  ggplot(aes(x = visit, y = sbp, group = pt_id, color = pt_id)) +
  geom_line(linewidth = 1) +
  geom_point(size = 3) +
  labs(title = "Systolic BP Across Visits",
       x = "Visit", y = "SBP (mmHg)", color = "Patient") +
  theme_classic()

Customizing Plots

patients %>%
  ggplot(aes(x = Diagnosis, y = HbA1c, fill = Diagnosis)) +
  geom_boxplot(alpha = 0.5) +
  scale_fill_brewer(palette = "Set2") +
  labs(title = "HbA1c Levels by Diagnosis",
       x = "Diagnosis", y = "HbA1c (%)") +
  theme(legend.position = "none") +
  theme_classic()

Key Themes

Theme Description
theme_classic() White background, axes only
theme_bw() White background + grid lines
theme_minimal() Minimal decoration
theme_light() Light tone
