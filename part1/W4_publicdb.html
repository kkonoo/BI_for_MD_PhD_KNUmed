1. Public Databases

Public databases are used for pilot studies or external validation.

Key Repositories (RNA-seq)

DB Data Provided Notes
GEO count matrix + metadata Most widely used
SRA raw FASTQ Requires SRA Toolkit
EGA raw FASTQ European; access request + fee required
TCGA/GDC multi-omics Cancer-specific
GTEx count matrix Normal tissue reference
Tip

You can find GSE/SRA accession numbers in the Data Availability section of publications.

Where we start

GEO and TCGA provide preprocessed count matrices, so you can skip the FASTQ → alignment step.

2. Browsing GEO

How to Search

① Boolean operators

  • Lung Cancer AND RNA-seq
  • NSCLC OR Adenocarcinoma
  • NOT Cell line

② MeSH Terms: “Lung Neoplasms”[MeSH Terms]

GEO Terminology

Prefix Meaning Example
GSE Series (study/paper level) GSE159857
GSM Sample (individual tissue) GSM4839001
GPL Platform (sequencing platform) GPL20301

Criteria for a Good Dataset

  • Sample size: sufficient N (minimum 30 recommended)
  • Experimental design: comparable control vs. case groups
  • Metadata: key variables available (e.g., duration, drug response, age, sex)

② Retrieving Expression Data

# Method 1: exprs() — often empty
expr_matrix <- exprs(gset_data)
dim(expr_matrix)

# Method 2: getGEOSuppFiles() — download supplementary files
getGEOSuppFiles(gse_id)

# Method 3: manual download (most reliable)
file_path <- 'data_w4/GSE159857_raw_counts_GRCh38.p13_NCBI.tsv.gz'
raw_counts <- read.table(file_path, header = TRUE, sep = "\t", row.names = 1)
head(raw_counts)

③ Matching Expression Data with Metadata (Important!)

Important

The row (sample) order in metadata must match the column (sample) order in the count matrix exactly.

# find common samples
common_samples <- intersect(colnames(raw_counts), clean_meta$geo_accession)
length(common_samples)

# align order
clean_meta <- clean_meta[match(common_samples, clean_meta$geo_accession), ]
raw_counts <- raw_counts[, match(common_samples, colnames(raw_counts))]

# verify
all(clean_meta$geo_accession == colnames(raw_counts))  # must be TRUE

# save
write.table(clean_meta, file = 'data_w4/clean_meta.txt',
            sep = "\t", quote = F, row.names = F, col.names = T)
write.table(raw_counts, file = 'data_w4/raw_counts.txt',
            sep = "\t", quote = F, row.names = F, col.names = T)

① Retrieving Clinical Data

# install packages
BiocManager::install("TCGAbiolinks")
BiocManager::install("SummarizedExperiment")

library(tidyverse)
library(TCGAbiolinks)
library(SummarizedExperiment)

# metadata
project_name <- "TCGA-LUAD"
clinical_luad <- GDCquery_clinic(project = project_name, type = "clinical") %>%
  as_tibble()

names(clinical_luad)

# select relevant columns
clinical_luad_subset <- clinical_luad %>%
  select(submitter_id, age_at_index, gender,
         vital_status, days_to_death, ajcc_pathologic_stage)

write.table(clinical_luad_subset, 'data_w4/clinical_luad_subset.txt',
            sep = "\t", quote = F, row.names = F, col.names = T)
saveRDS(clinical_luad_subset, 'data_w4/clinical_luad_subset.rds')

② Retrieving Expression Data

# set up query (may take time)
query_luad <- GDCquery(
  project = "TCGA-LUAD",
  data.category = "Transcriptome Profiling",
  experimental.strategy = "RNA-Seq",
  workflow.type = "STAR - Counts"  # select Raw Counts for DESeq2
)
saveRDS(query_luad, 'data_w4/query_luad.rds')

# (for practice) use only the first 20 samples
query_luad$results[[1]] <- query_luad$results[[1]] %>%
  filter(access == 'open') %>% .[1:20, ]

# download
GDCdownload(query_luad)

# prepare as SummarizedExperiment object
luad_data <- GDCprepare(query_luad)
class(luad_data)

③ Matching Count Matrix with Metadata

# Count Matrix
count_matrix <- assay(luad_data)
head(count_matrix)  # gene ID = Ensembl ID

# Metadata
meta_data <- colData(luad_data) %>% as_tibble
names(meta_data) %>% head

clean_meta <- meta_data %>%
  select(patient, sample, definition, tissue_type,
         ajcc_pathologic_stage, race, gender, age_at_index, days_to_birth)

# check
table(clean_meta$tissue_type)
table(clean_meta$definition)

Tips: Gene ID Conversion

Common gene ID systems used in RNA-seq data:

ID Type Example Source Use Case
Gene Symbol TP53, EGFR HGNC Human-readable (may have duplicates)
Ensembl ID ENSG00000141510 EBI RNA-seq aligner output; unique and stable
Entrez ID 7157 NCBI Gene set labels (KEGG, GO)

Conversion Method: org.Hs.eg.db + mapIds()

library(tidyverse)
library(org.Hs.eg.db)
library(AnnotationDbi)

# remove version numbers from Ensembl IDs
my_ids <- rownames(count_matrix)
head(my_ids)  # ENSG00000141510.15

my_ids <- sapply(my_ids, function(x) {
  strsplit(x, '\\.') %>% unlist %>% .[[1]]
})
my_ids <- as.character(my_ids)

# Ensembl → Symbol
my_symbol <- mapIds(org.Hs.eg.db,
                    keys = my_ids,
                    column = "SYMBOL",
                    keytype = "ENSEMBL",
                    multiVals = "first")

# Ensembl → Entrez ID (for GSEA/KEGG)
my_entrez <- mapIds(org.Hs.eg.db,
                    keys = my_ids,
                    column = "ENTREZID",
                    keytype = "ENSEMBL",
                    multiVals = "first")

# inspect
my_genes <- tibble(ids = my_ids,
                   symbol = my_symbol,
                   entrez = my_entrez)

# check for NAs (novel genes, pseudogenes, version mismatches, etc.)
sum(is.na(my_genes$symbol))
my_genes %>% filter(is.na(symbol))
